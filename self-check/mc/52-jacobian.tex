\section{Jacobian Matrix and Determinant}

\subsection{Basic Definitions}

\begin{enumerate}
  \item Let $F : \mathbb{R}^{n} \to \mathbb{R}^{m}$ be given by
    \[
      F(x_{1},\dots,x_{n}) = (f_{1}(x),\dots,f_{m}(x)).
    \]
    What is the definition of the \emph{Jacobian matrix} $J_{F}(x)$?
  \item Write explicitly the Jacobian matrix:
    \[
      J_{F}(x) = \left( \frac{\partial f_{i}}{\partial x_{j}}\right).
    \]
    What are its dimensions?
  \item In which case is the Jacobian matrix square?
  \item When the Jacobian matrix is simply the transpose of the gradient vector?
  \item When $m = n$, how is the \emph{Jacobian determinant} defined?
  \item What does the notation
    \[
      \det J_{F}(x)
    \]
    represent?
\end{enumerate}

\subsection{Jacobian as Derivative}

\begin{enumerate}[resume]
  \item How is the Jacobian matrix related to the total derivative (Fréchet derivative) of a function $F : \mathbb{R}^{n} \to \mathbb{R}^{m}$?
  \item State the linear approximation formula:
    \[
      F(x+h) \approx F(x) + J_{F}(x)h.
    \]
  \item What does the Jacobian matrix represent geometrically at a point?
  \item What condition on $J_{F}(x)$ ensures that $F$ is locally invertible near $x$?
\end{enumerate}

\subsection{Chain Rule}

\begin{enumerate}[resume]
  \item Let $F : \mathbb{R}^{n} \to \mathbb{R}^{m}$ and $G : \mathbb{R}^{m} \to \mathbb{R}^{p}$. State the multivariable chain rule for Jacobians:
    \[
      J_{G \circ F}(x) = ?
    \]
  \item When $n=m=p$, what is the corresponding formula for determinants?
  \item Why does this imply
    \[
      \det J_{G \circ F}(x) = \det J_{G}(F(x)) \cdot \det J_{F}(x)?
    \]
\end{enumerate}

\subsection{Geometric Interpretation}

\begin{enumerate}[resume]
  \item How does the Jacobian determinant describe local volume scaling?
  \item What does it mean if $\det J_{F}(x) > 0$?
  \item What does it mean if $\det J_{F}(x) < 0$?
  \item What does it mean if $\det J_{F}(x) = 0$?
  \item How does the absolute value $|\det J_{F}(x)|$ relate to infinitesimal volume elements?
\end{enumerate}

\subsection{Change of Variables in Integration}

\begin{enumerate}[resume]
  \item State the change-of-variables formula in $\mathbb{R}^{n}$:
    \[
      \int_{F(U)}g(y)\,dy = \int_{U} g(F(x))\, |\det J_{F}(x)| \, dx.
    \]
  \item Why is the absolute value necessary?
  \item What conditions must $F$ satisfy for the change-of-variables formula to hold?
  \item Compute the Jacobian determinant for the transformation from Cartesian to polar coordinates:
    \[
      x = r\cos\theta, \quad y = r\sin\theta.
    \]
  \item Why does the area element become
    \[
      dx\,dy = r\,dr\,d\theta?
    \]
\end{enumerate}

\subsection{Special Cases and Examples}

\begin{enumerate}[resume]
  \item Compute the Jacobian matrix of
    \[
      F(x,y) = (x^{2} y, e^{xy}).
    \]
  \item Compute the Jacobian determinant of
    \[
      F(x,y) = (u,v)
    \]
    where
    \[
      u = x+y, \quad v = x-y.
    \]
  \item For a linear transformation $F(x)=Ax$, what is $J_{F}(x)$?
  \item How is $\det J_{F}(x)$ related to $\det A$ in the linear case?
  \item If $F$ is a rotation in $\mathbb{R}^{2}$, what is its Jacobian determinant?
\end{enumerate}

\subsection{Inverse Function Theorem}

\begin{enumerate}[resume]
  \item State the inverse function theorem in terms of the Jacobian determinant.
  \item Why does $\det J_{F}(x_{0}) \neq 0$ imply local invertibility?
  \item How is the Jacobian of the inverse function expressed in terms of $J_{F}$?
\end{enumerate}
