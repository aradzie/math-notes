For the following exercises, find the derivative $\frac{dy}{dx}$.

\begin{enumerate}[start=1]
\item $y=\ln(2x)$
\item $y=\ln(2x+1)$
\item $y=\frac{1}{\ln x}$
\end{enumerate}

For the following exercises, find the indefinite integral.

\begin{enumerate}[resume]
\item $\int \frac{dt}{3t}$
\item $\int \frac{dx}{1+x}$
\end{enumerate}

For the following exercises, find the derivative $dy/dx$. (You can use a calculator to plot the function and the derivative to confirm that it is correct.)

\begin{enumerate}[resume]
\item \techrequired $y=\frac{\ln(x)}{x}$
\item \techrequired $y=x\ln(x)$
\item \techrequired $y=\log_{10}x$
\item \techrequired $y=\ln(\sin x)$
\item \techrequired $y=\ln(\ln x)$
\item \techrequired $y=7\ln(4x)$
\item \techrequired $y=\ln({(4x)}^{7})$
\item \techrequired $y=\ln(\tan x)$
\item \techrequired $y=\ln(\tan(3x))$
\item \techrequired $y=\ln(\cos^{2}x)$
\end{enumerate}

For the following exercises, find the definite or indefinite integral.

\begin{enumerate}[resume]
\item $\int_{0}^{1}\frac{dx}{3+x}$
\item $\int_{0}^{1}\frac{dt}{3+2t}$
\item $\int_{0}^{2}\frac{x\,dx}{x^{2}+1}$
\item $\int_{0}^{2}\frac{x^{3}\,dx}{x^{2}+1}$
\item $\int_{2}^{e}\frac{dx}{x\ln x}$
\item $\int_{2}^{e}\frac{dx}{x\,{(\ln x)}^{2}}$
\item $\int \frac{\cos x\,dx}{\sin x}$
\item $\int_{0}^{\pi/4}\tan x\,dx$
\item $\int \cot(3x)\,dx$
\item $\int \frac{{(\ln x)}^{2}\,dx}{x}$
\end{enumerate}

For the following exercises, compute $dy/dx$ by differentiating $\ln y$.

\begin{enumerate}[resume]
\item $y=\sqrt{x^{2}+1}$
\item $y=\sqrt{x^{2}+1}\sqrt{x^{2}-1}$
\item $y=e^{\sin x}$
\item $y=x^{-1/x}$
\item $y=e^{(ex)}$
\item $y=x^{e}$
\item $y=x^{(ex)}$
\item $y=\sqrt{x}\,\sqrt[3]{x}\,\sqrt[6]{x}$
\item $y=x^{-1/\ln x}$
\item $y=e^{-\ln x}$
\end{enumerate}

For the following exercises, evaluate by any method.

\begin{enumerate}[resume]
\item $\int_{5}^{10}\frac{dt}{t}-\int_{5x}^{10x}\frac{dt}{t}$
\item $\int_{1}^{e^{\pi}}\frac{dx}{x}+\int_{-2}^{-1}\frac{dx}{x}$
\item $\frac{d}{dx}\int_{x}^{1}\frac{dt}{t}$
\item $\frac{d}{dx}\int_{x}^{x^{2}}\frac{dt}{t}$
\item $\frac{d}{dx}\ln(\sec x+\tan x)$
\end{enumerate}

For the following exercises, verify the derivatives and antiderivatives.

\begin{enumerate}[resume]
\item $\frac{d}{dx}\ln(x+\sqrt{x^{2}+1})=\frac{1}{\sqrt{1+x^{2}}}$
\item $\frac{d}{dx}\ln(\frac{x-a}{x+a})=\frac{2a}{(x^{2}-a^{2})}$
\item $\frac{d}{dx}\ln(\frac{1+\sqrt{1-x^{2}}}{x})=-\frac{1}{x\sqrt{1-x^{2}}}$
\item $\frac{d}{dx}\ln(x+\sqrt{x^{2}-a^{2}})=\frac{1}{\sqrt{x^{2}-a^{2}}}$
\item $\int \frac{dx}{x\ln(x)\ln(\ln x)}=\ln(\ln(\ln x))+C$
\end{enumerate}
