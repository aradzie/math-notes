For the following exercises, evaluate each function at the indicated values.

\begin{enumerate}[start=1]
\item $W(x,y)=4x^{2}+y^{2}$. Find $W(2,-1)$, $W(-3,6)$.
\item $W(x,y)=4x^{2}+y^{2}$. Find $W(2+h,3+h)$.
\item The volume of a right circular cylinder is calculated by a function of two variables, $V(x,y)=\pi x^{2}y$, where $x$ is the radius of the right circular cylinder and $y$ represents the height of the cylinder. Evaluate $V(2,5)$ and explain what this means.
\item An oxygen tank is constructed of a right cylinder of height $y$ and radius $x$ with two hemispheres of radius $x$ mounted on the top and bottom of the cylinder. Express the volume of the tank as a function of two variables, $x$ and $y$, find $V(10,2)$, and explain what this means.
\end{enumerate}

For the following exercises, find the domain of the function.

\begin{enumerate}[resume]
\item $V(x,y)=4x^{2}+y^{2}$
\item $f(x,y)=\sqrt{x^{2}+y^{2}-4}$
\item $f(x,y)=4\ln(y^{2}-x)$
\item $g(x,y)=\sqrt{16-4x^{2}-y^{2}}$
\item $z(x,y)=y^{2}-x^{2}$
\item $f(x,y)=\frac{y+2}{x^{2}}$
\end{enumerate}

Find the range of the functions.

\begin{enumerate}[resume]
\item $g(x,y)=\sqrt{16-4x^{2}-y^{2}}$
\item $V(x,y)=4x^{2}+y^{2}$
\item $z=y^{2}-x^{2}$
\end{enumerate}

For the following exercises, find the level curves of each function at the indicated value of $c$ to visualize the given function.

\begin{enumerate}[resume]
\item $z(x,y)=y^{2}-x^{2}$, $c=1$
\item $z(x,y)=y^{2}-x^{2}$, $c=4$
\item $g(x,y)=x^{2}+y^{2}$, $c=4,c=9$
\item $g(x,y)=4-x-y$, $c=0,4$
\item $f(x,y)=xy$, $c=1$, $c=-1$
\item $h(x,y)=2x-y$, $c=0,-2,2$
\item $f(x,y)=x^{2}-y$, $c=1,2$
\item $g(x,y)=\frac{x}{x+y}$, $c=-1,0,2$
\item $g(x,y)=x^{3}-y$, $c=-1,0,2$
\item $g(x,y)=e^{xy}$, $c=\frac{1}{2},3$
\item $f(x,y)=x^{2}$, $c=4,9$
\item $f(x,y)=xy-x$, $c=-2,0,2$
\item $h(x,y)=\ln(x^{2}+y^{2})$, $c=-1,0,1$
\item $g(x,y)=\ln(\frac{y}{x^{2}})$, $c=-2,0,2$
\item $z=f(x,y)=\sqrt{x^{2}+y^{2}}$, $c=3$
\item $f(x,y)=\frac{y+2}{x^{2}}$, $c=$ any constant
\end{enumerate}

For the following exercises, find the vertical traces of the functions at the indicated values of $x$ and $y$, and plot the traces.

\begin{enumerate}[resume]
\item $z=4-x-y$, $x=2$
\item $f(x,y)=3x+y^{3}$, $x=1$
\item $z=\cos \sqrt{x^{2}+y^{2}}$, $x=1$
\end{enumerate}

Find the domain of the following functions.

\begin{enumerate}[resume]
\item $z=\sqrt{100-4x^{2}-25y^{2}}$
\item $z=\ln(x-y^{2})$
\item $f(x,y,z)=\frac{1}{\sqrt{36-4x^{2}-9y^{2}-z^{2}}}$
\item $f(x,y,z)=\sqrt{49-x^{2}-y^{2}-z^{2}}$
\item $f(x,y,z)=\sqrt[3]{16-x^{2}-y^{2}-z^{2}}$
\item $f(x,y)=\cos \sqrt{x^{2}+y^{2}}$
\end{enumerate}

For the following exercises, plot a graph of the function.

\begin{enumerate}[resume]
\item $z=f(x,y)=\sqrt{x^{2}+y^{2}}$
\item $z=x^{2}+y^{2}$
\item Use technology to graph $z=x^{2}y$.
\end{enumerate}

Sketch the following by finding the level curves. Verify the graph using technology.

\begin{enumerate}[resume]
\item $f(x,y)=\sqrt{4-x^{2}-y^{2}}$
\item $f(x,y)=2-\sqrt{x^{2}+y^{2}}$
\item $z=1+e^{-x^{2}-y^{2}}$
\item $z=\cos \sqrt{x^{2}+y^{2}}$
\item $z=y^{2}-x^{2}$
\item Describe the contour lines for several values of $c$ for $z=x^{2}+y^{2}-2x-2y$.
\end{enumerate}

Find the level surface for the functions of three variables and describe it.

\begin{enumerate}[resume]
\item $w(x,y,z)=x-2y+z$, $c=4$
\item $w(x,y,z)=x^{2}+y^{2}+z^{2}$, $c=9$
\item $w(x,y,z)=x^{2}+y^{2}-z^{2}$, $c=-4$
\item $w(x,y,z)=x^{2}+y^{2}-z^{2}$, $c=4$
\item $w(x,y,z)=9x^{2}-4y^{2}+36z^{2}$, $c=0$
\end{enumerate}

For the following exercises, find an equation of the level curve of $f$ that contains the point $P$.

\begin{enumerate}[resume]
\item $f(x,y)=1-4x^{2}-y^{2}$, $P(0,1)$
\item $g(x,y)=y^{2}\arctan x$, $P(1,2)$
\item $g(x,y)=e^{xy}(x^{2}+y^{2})$, $P(1,0)$
\item The strength $E$ of an electric field at point $(x,y,z)$ resulting from an infinitely long charged wire lying along the $z$-axis is given by $E(x,y,z)=k/\sqrt{x^{2}+y^{2}}$, where $k$ is a positive constant. For simplicity, let $k=1$ and find the equations of the level surfaces for $E=10$ and $E=100$.
\item A thin plate made of iron is located in the $xy$-plane. The temperature $T$ in degrees Celsius at a point $P(x,y)$ is inversely proportional to the square of its distance from the origin. Express $T$ as a function of $x$ and $y$.
\item Refer to the preceding problem. Using the temperature function found there, determine the proportionality constant if the temperature at point $P(1,2)$ is $50^\circ \text{C}$. Use this constant to determine the temperature at point $Q(3,4)$.
\item Refer to the preceding problem. Find the level curves for $T=40^\circ \text{C}$ and $T=100^\circ \text{C}$, and describe what the level curves represent.
\end{enumerate}
