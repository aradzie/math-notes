Determine the order of the following differential equations.

\begin{enumerate}[start=1]
\item $y'+y=3y^{2}$
\item ${(y')}^{2}=y'+2y$
\item $y'''+y''y'=3x^{2}$
\item $y'=y''+3t^{2}$
\item $\frac{dy}{dt}=t$
\item $\frac{dy}{dx}+\frac{d^{2}y}{dx^{2}}=3x^{4}$
\item ${(\frac{dy}{dt})}^{2}+8\frac{dy}{dt}+3y=4t$
\end{enumerate}

Verify that the following functions are solutions to the given differential equation.

\begin{enumerate}[resume]
\item $y=\frac{x^{3}}{3}$ solves $y'=x^{2}$
\item $y=2e^{-x}+x-1$ solves $y'=x-y$
\item $y=e^{3x}-\frac{e^{x}}{2}$ solves $y'=3y+e^{x}$
\item $y=\frac{1}{1-x}$ solves $y'=y^{2}$
\item $y=e^{x^{2}/2}$ solves $y'=xy$
\item $y=4+\ln x$ solves $xy'=1$
\item $y=3-x+x\ln x$ solves $y'=\ln x$
\item $y=2e^{x}-x-1$ solves $y'=y+x$
\item $y=e^{x}+\frac{\sin x}{2}-\frac{\cos x}{2}$ solves $y'=\cos x+y$
\item $y=\pi e^{-\cos x}$ solves $y'=y\sin x$
\end{enumerate}

Verify the following general solutions and find the particular solution.

\begin{enumerate}[resume]
\item Find the particular solution to the differential equation $y'=4x^{2}$ that passes through $(-3,-30)$, given that $y=C+\frac{4x^{3}}{3}$ is a general solution.
\item Find the particular solution to the differential equation $y'=3x^{3}$ that passes through $(1,4.75)$, given that $y=C+\frac{3x^{4}}{4}$ is a general solution.
\item Find the particular solution to the differential equation $y'=3x^{2}y$ that passes through $(0,12)$, given that $y=Ce^{x^{3}}$ is a general solution.
\item Find the particular solution to the differential equation $y'=2xy$ that passes through $(0,\frac{1}{2})$, given that $y=Ce^{x^{2}}$ is a general solution.
\item Find the particular solution to the differential equation $y'={(2xy)}^{2}$ that passes through $(1,-\frac{1}{2})$, given that $y=-\frac{3}{C+4x^{3}}$ is a general solution.
\item Find the particular solution to the differential equation $y'x^{2}=y$ that passes through $(1,\frac{2}{e})$, given that $y=Ce^{-1/x}$ is a general solution.
\item Find the particular solution to the differential equation $8\frac{dx}{dt}=-2\cos(2t)-\cos(4t)$ that passes through $(\pi,\pi)$, given that $x=C-\frac{1}{8}\sin(2t)-\frac{1}{32}\sin(4t)$ is a general solution.
\item Find the particular solution to the differential equation $\frac{du}{dt}=\tan u$ that passes through $(1,\frac{\pi}{2})$, given that $u=\sin^{-1}(e^{C+t})$ is a general solution.
\item Find the particular solution to the differential equation $\frac{dy}{dt}=e^{(t+y)}$ that passes through $(1,0)$, given that $y=-\ln(C-e^{t})$ is a general solution.
\item Find the particular solution to the differential equation $y'(1-x^{2})=1+y$ that passes through $(0,-2)$, given that $y=C\frac{\sqrt{x+1}}{\sqrt{1-x}}-1$ is a general solution.
\end{enumerate}

\begin{enumerate}[resume]

\item $y'=\ln x+\tan x$
\item $y'=\sin xe^{\cos x}$

Solve the following initial-value problems starting from $y(0)=1$ and $y(0)=-1$. Draw both solutions on the same graph.

\begin{enumerate}[resume]
\item $\frac{dy}{dt}=2t$
\item $\frac{dy}{dt}=-t$
\item $\frac{dy}{dt}=2y$
\item $\frac{dy}{dt}=-y$
\item $\frac{dy}{dt}=2$
\end{enumerate}

Solve the following initial-value problems starting from $y_{0}=10$. At what time does $y$ increase to $100$ or drop to $1$?

\begin{enumerate}[resume]
\item $\frac{dy}{dt}=4t$
\item $\frac{dy}{dt}=4y$
\item $\frac{dy}{dt}=-2y$
\item $\frac{dy}{dt}=e^{4t}$
\item $\frac{dy}{dt}=e^{-4t}$
\end{enumerate}
