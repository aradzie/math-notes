For the following exercises, calculate the partial derivative using the limit definitions only.

\begin{enumerate}[start=1]
\item $\frac{\partial z}{\partial x}$ for $z=x^{2}-3xy+y^{2}$
\item $\frac{\partial z}{\partial y}$ for $z=x^{2}-3xy+y^{2}$
\end{enumerate}

For the following exercises, calculate the partial derivatives.

\begin{enumerate}[resume]
\item $\frac{\partial z}{\partial x}$ for $z=\sin(3x)\cos(3y)$
\item $\frac{\partial z}{\partial y}$ for $z=\sin(3x)\cos(3y)$
\item $\frac{\partial z}{\partial x}$ and $\frac{\partial z}{\partial y}$ for $z=x^{8}e^{3y}$
\item $\frac{\partial z}{\partial x}$ and $\frac{\partial z}{\partial y}$ for $z=\ln(x^{6}+y^{4})$
\item Find $f_{y}(x,y)$ for $f(x,y)=e^{xy}\cos(x)\sin(y)$.
\item Let $z=e^{xy}$. Find $\frac{\partial z}{\partial x}$ and $\frac{\partial z}{\partial y}$.
\item Let $z=\ln(\frac{x}{y})$. Find $\frac{\partial z}{\partial x}$ and $\frac{\partial z}{\partial y}$.
\item Let $z=\tan(2x-y)$. Find $\frac{\partial z}{\partial x}$ and $\frac{\partial z}{\partial y}$.
\item Let $z=\sinh(2x+3y)$. Find $\frac{\partial z}{\partial x}$ and $\frac{\partial z}{\partial y}$.
\item Let $f(x,y)=\arctan(\frac{y}{x})$. Evaluate $f_{x}(2,-2)$ and $f_{y}(2,-2)$.
\item Let $f(x,y)=\frac{xy}{x-y}$. Find $f_{x}(2,-2)$ and $f_{y}(2,-2)$.
\item Find $\frac{\partial z}{\partial x}$ at $(0,1)$ for $z=e^{-x}\cos(y)$. Evaluate the partial derivatives at point $P(0,1)$.
\item Given $f(x,y,z)=x^{3}yz^{2}$, find $\frac{\partial^{2}f}{\partial x\partial y}$ and $f_{z}(1,1,1)$.
\item Given $f(x,y,z)=2\sin(x+y)$, find $f_{x}(0,\frac{\pi}{2},-4)$, $f_{y}(0,\frac{\pi}{2},-4)$, and $f_{z}(0,\frac{\pi}{2},-4)$.
\item The area of a parallelogram with adjacent side lengths that are $a$ and $b$, and in which the angle between these two sides is $\theta$, is given by the function $A(a,b,\theta)=ba\sin(\theta)$. Find the rate of change of the area of the parallelogram with respect to the following:
\begin{enumerate}[label=(\alph*)]
  \item Side $a$
  \item Side $b$
  \item Angle $\theta$
\end{enumerate}
\item Express the volume of a right circular cylinder as a function of two variables:
\begin{enumerate}[label=(\alph*)]
  \item its radius $r$ and its height $h$.
  \item Show that the rate of change of the volume of the cylinder with respect to its radius is the product of its circumference multiplied by its height.
  \item Show that the rate of change of the volume of the cylinder with respect to its height is equal to the area of the circular base.
\end{enumerate}
\item Calculate $\frac{\partial w}{\partial z}$ for $w=z\sin(xy^{2}+2z)$.
\end{enumerate}

Find the indicated higher-order partial derivatives.

\begin{enumerate}[resume]
\item $f_{xy}$ for $z=\ln(x-y)$
\item $f_{yx}$ for $z=\ln(x-y)$
\item Let $z=x^{2}+3xy+2y^{2}$. Find $\frac{\partial^{2}z}{\partial x^{2}}$ and $\frac{\partial^{2}z}{\partial y^{2}}$.
\item Given $z=e^{x}\tan y$, find $\frac{\partial^{2}z}{\partial x\partial y}$ and $\frac{\partial^{2}z}{\partial y\partial x}$.
\item Given $f(x,y,z)=xyz$, find $f_{xyy},f_{yxy}$, and $f_{yyx}$.
\item Given $f(x,y,z)=e^{-2x}\sin(z^{2}y)$, show that $f_{xyy}=f_{yxy}$.
\item Show that $z=\frac{1}{2}(e^{y}-e^{-y})\sin x$ is a solution of the differential equation $\frac{\partial^{2}z}{\partial x^{2}}+\frac{\partial^{2}z}{\partial y^{2}}=0$.
\item Find $f_{xx}(x,y)$ for $f(x,y)=\frac{4x^{2}}{y}+\frac{y^{2}}{2x}$.
\item Let $f(x,y,z)=x^{2}y^{3}z-3xy^{2}z^{3}+5x^{2}z-y^{3}z$. Find $f_{xyz}$.
\item Let $F(x,y,z)=x^{3}yz^{2}-2x^{2}yz+3xz-2y^{3}z$. Find $F_{xyz}$.
\item Given $f(x,y)=x^{2}+x-3xy+y^{3}-5$, find all points at which $f_{x}=f_{y}=0$ simultaneously.
\item Given $f(x,y)=2x^{2}+2xy+y^{2}+2x-3$, find all points at which $\frac{\partial f}{\partial x}=0$ and $\frac{\partial f}{\partial y}=0$ simultaneously.
\item Given $f(x,y)=y^{3}-3yx^{2}-3y^{2}-3x^{2}+1$, find all points on $f$ at which $f_{x}=f_{y}=0$ simultaneously.
\item Given $f(x,y)=15x^{3}-3xy+15y^{3}$, find all points at which $f_{x}(x,y)=f_{y}(x,y)=0$ simultaneously.
\item Show that $z=e^{x}\sin y$ satisfies the equation $\frac{\partial^{2}z}{\partial x^{2}}+\frac{\partial^{2}z}{\partial y^{2}}=0$.
\item Show that $f(x,y)=\ln(x^{2}+y^{2})$ solves Laplace's equation $\frac{\partial^{2}z}{\partial x^{2}}+\frac{\partial^{2}z}{\partial y^{2}}=0$.
\item Show that $z=e^{-t}\cos(\frac{x}{c})$ satisfies the heat equation $\frac{\partial z}{\partial t}=c^{2}\left( \frac{\partial^{2}z}{\partial x^{2}} \right)$.
\item Find $\lim_{\Delta x\to0}\frac{f(x+\Delta x,\,y)-f(x,y)}{\Delta x}$ for $f(x,y)=-7x-2xy+7y$.
\item Find $\lim_{\Delta y\to0}\frac{f(x,y+\Delta y)-f(x,y)}{\Delta y}$ for $f(x,y)=-7x-2xy+7y$.
\item Find $\lim_{\Delta x\to0}\frac{\Delta f}{\Delta x}=\lim_{\Delta x\to0}\frac{f(x+\Delta x,y)-f(x,y)}{\Delta x}$ for $f(x,y)=x^{2}y^{2}+xy+y$.
\item Find $\lim_{\Delta x\to0}\frac{\Delta f}{\Delta x}=\lim_{\Delta x\to0}\frac{f(x+\Delta x,y)-f(x,y)}{\Delta x}$ for $f(x,y)=\sin(xy)$.
\item The function $P(T,V)=\frac{nRT}{V}$ gives the pressure at a point in a gas as a function of temperature $T$ and volume $V$. The letters $n$ and $R$ are constants. Find $\frac{\partial P}{\partial V}$ and $\frac{\partial P}{\partial T}$, and explain what these quantities represent.
\item The equation for heat flow in the $xy$-plane is $\frac{\partial f}{\partial t}=\frac{\partial^{2}f}{\partial x^{2}}+\frac{\partial^{2}f}{\partial y^{2}}$. Show that $f(x,y,t)=e^{-2t}\sin x\sin y$ is a solution.
\item The basic wave equation is $f_{tt}=f_{xx}$. Verify that $f(x,t)=\sin(x+t)$ and $f(x,t)=\sin(x-t)$ are solutions.
\item The law of cosines can be thought of as a function of three variables. Let $x,y$, and $\theta$ be two sides of any triangle where the angle $\theta$ is the included angle between the two sides. Then, $F(x,y,\theta)=x^{2}+y^{2}-2xy\cos \theta$ gives the square of the third side of the triangle. Find $\frac{\partial F}{\partial \theta}$ and $\frac{\partial F}{\partial x}$ when $x=2,y=3$, and $\theta=\frac{\pi}{6}$.
\item Suppose the sides of a rectangle are changing with respect to time. The first side is changing at a rate of $2$ in./sec whereas the second side is changing at the rate of $4$ in/sec. How fast is the diagonal of the rectangle changing when the first side measures $16$ in. and the second side measures $20$ in.? (Round answer to three decimal places.)
\item A Cobb-Douglas production function is $f(x,y)=200x^{0.7}y^{0.3}$, where $x$ and $y$ represent the amount of labor and capital available. Let $x=500$ and $y=1000$. Find $\frac{\partial f}{\partial x}$ and $\frac{\partial f}{\partial y}$ at these values, which represent the marginal productivity of labor and capital, respectively.
\item The apparent temperature index is a measure of how the temperature feels, and it is based on two variables: $h$, which is relative humidity, and $t$, which is the air temperature. $A=0.885t-22.4h+1.20th-0.544$. Find $\frac{\partial A}{\partial t}$ and $\frac{\partial A}{\partial h}$ when $t=20^\circ \text{F}$ and $h=0.90$.
\end{enumerate}
