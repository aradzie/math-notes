Classify each of the following equations as linear or nonlinear. If the equation is linear, determine whether it is homogeneous or nonhomogeneous.

\begin{enumerate}[start=1]
\item $x^{3}y''+(x-1)y'-8y=0$
\item $(1+y^{2})y''+xy'-3y=\cos x$
\item $xy''+e^{y}y'=x$
\item $y''+\frac{4}{x}y'-8xy=5x^{2}+1$
\item $y''+(\sin x)y'-xy=4y$
\item $y''+(\frac{x+3}{y})y'=0$
\end{enumerate}

For each of the following problems, verify that the given function is a solution to the differential equation. Use a graphing utility to graph the particular solutions for several values of $c_{1}$ and $c_{2}$. What do the solutions have in common?

\begin{enumerate}[resume]
\item \techrequired$y''+2y'-3y=0$; $y(x)=c_{1}e^{x}+c_{2}e^{-3x}$
\item \techrequired$x^{2}y''-2y-3x^{2}+1=0$; $y(x)=c_{1}x^{2}+c_{2}x^{-1}+x^{2}\ln (x)+\frac{1}{2}$
\item \techrequired$y''+14y'+49y=0$; $y(x)=c_{1}e^{-7x}+c_{2}xe^{-7x}$
\item \techrequired$6y''-49y'+8y=0$; $y(x)=c_{1}e^{x/6}+c_{2}e^{8x}$
\end{enumerate}

Find the general solution to the linear differential equation.

\begin{enumerate}[resume]
\item $y''-3y'-10y=0$
\item $y''-7y'+12y=0$
\item $y''+4y'+4y=0$
\item $4y''-12y'+9y=0$
\item $2y''-3y'-5y=0$
\item $3y''-14y'+8y=0$
\item $y''+y'+y=0$
\item $5y''+2y'+4y=0$
\item $y''-121y=0$
\item $8y''+14y'-15y=0$
\item $y''+81y=0$
\item $y''-y'+11y=0$
\item $2y''=0$
\item $y''-6y'+9y=0$
\item $3y''-2y'-7y=0$
\item $4y''-10y'=0$
\item $36\frac{d^{2}y}{dx^{2}}+12\frac{dy}{dx}+y=0$
\item $25\frac{d^{2}y}{dx^{2}}-80\frac{dy}{dx}+64y=0$
\item $\frac{d^{2}y}{dx^{2}}-9\frac{dy}{dx}=0$
\item $4\frac{d^{2}y}{dx^{2}}+8y=0$
\end{enumerate}

Solve the initial-value problem.

\begin{enumerate}[resume]
\item $y''+5y'+6y=0$, $y(0)=0$, $y'(0)=-2$
\item $y''+2y'-8y=0$, $y(0)=5$, $y'(0)=4$
\item $y''+4y=0$, $y(0)=3$, $y'(0)=10$
\item $y''-18y'+81y=0$, $y(0)=1$, $y'(0)=5$
\item $y''-y'-30y=0$, $y(0)=1$, $y'(0)=-16$
\item $4y''+4y'-8y=0$, $y(0)=2$, $y'(0)=1$
\item $25y''+10y'+y=0$, $y(0)=2$, $y'(0)=1$
\item $y''+y=0$, $y(\pi)=1$, $y'(\pi)=-5$
\end{enumerate}

Solve the boundary-value problem, if possible.

\begin{enumerate}[resume]
\item $y''+y'-42y=0$, $y(0)=0$, $y(1)=2$
\item $9y''+y=0$, $y(\frac{3\pi}{2})=6$, $y(0)=-8$
\item $y''+10y'+34y=0$, $y(0)=6$, $y(\pi)=2$
\item $y''+7y'-60y=0$, $y(0)=4$, $y(2)=0$
\item $y''-4y'+4y=0$, $y(0)=2$, $y(1)=-1$
\item $y''-5y'=0$, $y(0)=3$, $y(-1)=2$
\item $y''+9y=0$, $y(0)=4$, $y(\frac{\pi}{3})=-4$
\item $4y''+25y=0$, $y(0)=2$, $y(2\pi)=-2$
\item Find a differential equation with a general solution that is $y=c_{1}e^{x/5}+c_{2}e^{-4x}$.
\item Find a differential equation with a general solution that is $y=c_{1}e^{x}+c_{2}e^{-4x/3}$.
\end{enumerate}

For each of the following differential equations:


(a) Solve the initial value problem.


(b) \techrequired Use a graphing utility to graph the particular solution.

\begin{enumerate}[resume]
\item $y''+64y=0$; $y(0)=3$, $y'(0)=-24$
\item $y''-2y'+10y=0$, $y(0)=1$, $y'(0)=13$
\item $y''+5y'+15y=0$, $y(0)=-2$, $y'(0)=7$
\item (Principle of superposition) Prove that if $y_{1}(x)$ and $y_{2}(x)$ are solutions to a linear homogeneous differential equation, $y''+p(x)y'+q(x)y=0$, then the function $y(x)=c_{1}y_{1}(x)+c_{2}y_{2}(x)$, where $c_{1}$ and $c_{2}$ are constants, is also a solution.
\item Prove that if $a$, $b$, and $c$ are positive constants, then all solutions to the second-order linear differential equation $ay''+by'+cy=0$ approach zero as $x\to \infty$. (Hint: Consider three cases: two distinct roots, repeated real roots, and complex conjugate roots.)
\end{enumerate}
