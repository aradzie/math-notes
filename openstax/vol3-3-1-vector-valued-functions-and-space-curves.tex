\begin{enumerate}[start=1]
\item Give the component functions $x=f(t)$ and $y=g(t)$ for the vector-valued function $\mathbf{r}(t)=3\sec t\,\mathbf{i}+2\tan t\,\mathbf{j}$.
\item Given $\mathbf{r}(t)=3\sec t\,\mathbf{i}+2\tan t\,\mathbf{j}$, find the following values (if possible).
\begin{enumerate}[label=(\alph*)]
  \item $\mathbf{r}(\frac{\pi}{4})$
  \item $\mathbf{r}(\pi)$
  \item $\mathbf{r}(\frac{\pi}{2})$
\end{enumerate}
\item Sketch the curve of the vector-valued function $\mathbf{r}(t)=3\sec t\,\mathbf{i}+2\tan t\,\mathbf{j}$ and give the orientation of the curve. Sketch asymptotes as a guide to the graph.
\item Evaluate $\lim_{t\to0}\langle e^{t}\,,\frac{\sin t}{t}\,,e^{-t}\rangle$.
\item Given the vector-valued function $\mathbf{r}(t)=\langle \cos t,\sin t\rangle$, find the following values:
\begin{enumerate}[label=(\alph*)]
  \item $\lim_{t\to \frac{\pi}{3}}\,\mathbf{r}(t)$
  \item $\mathbf{r}(\frac{\pi}{3})$
  \item Is $\mathbf{r}(t)$ continuous at $t=\frac{\pi}{3}$?
  \item Graph $\mathbf{r}(t)$.
\end{enumerate}
\item Given the vector-valued function $\mathbf{r}(t)=\langle t,t^{2}+1\rangle$, find the following values:
\begin{enumerate}[label=(\alph*)]
  \item $\lim_{t\to-3}\,\mathbf{r}(t)$
  \item $\mathbf{r}(-3)$
  \item Is $\mathbf{r}(t)$ continuous at $t=-3$?
  \item $\mathbf{r}(t+2)-\mathbf{r}(t)$
\end{enumerate}
\item Let $\mathbf{r}(t)=e^{t}\,\mathbf{i}+\sin t\,\mathbf{j}+\ln t\,\mathbf{k}$. Find the following values:
\begin{enumerate}[label=(\alph*)]
  \item $\mathbf{r}(\frac{\pi}{4})$
  \item $\lim_{t\to \pi/4}\,\mathbf{r}(t)$
  \item Is $\mathbf{r}(t)$ continuous at $t=\frac{\pi}{4}$?
\end{enumerate}
\end{enumerate}

Find the limit of the following vector-valued functions at the indicated value of $t$.

\begin{enumerate}[resume]
\item $\lim_{t\to4}\langle \sqrt{t-3},\frac{\sqrt{t}-2}{t-4},\tan(\frac{\pi}{t})\rangle$
\item $\lim_{t\to \pi/2}\,\mathbf{r}(t)$ for $\mathbf{r}(t)=e^{t}\,\mathbf{i}+\sin t\,\mathbf{j}+\ln t\,\mathbf{k}$
\item $\lim_{t\to \infty}\langle e^{-2t},\frac{2t+3}{3t-1},\arctan(2t)\rangle$
\item $\lim_{t\to e^{2}}\langle t\ln(t),\frac{\ln t}{t^{2}},\sqrt{\ln (t^{2})}\rangle$
\item $\lim_{t\to \pi/6}\langle \cos^{2}t,\sin^{2}t,1\rangle$
\item $\lim_{t\to \infty}\,\mathbf{r}(t)$ for $\mathbf{r}(t)=2e^{-t}\,\mathbf{i}+e^{-t}\,\mathbf{j}+\ln(t-1)\,\mathbf{k}$
\item Describe the curve defined by the vector-valued function $\mathbf{r}(t)=(1+t)\,\mathbf{i}+(2+5t)\,\mathbf{j}+(-1+6t)\,\mathbf{k}$.
\end{enumerate}

Find the domain of the vector-valued functions.

\begin{enumerate}[resume]
\item Domain: $\mathbf{r}(t)=\langle t^{2},\tan t,\ln t\rangle$
\item Domain: $\mathbf{r}(t)=\langle t^{2},\sqrt{t-3},\frac{3}{2t+1}\rangle$
\item Domain: $\mathbf{r}(t)=\langle \csc(t),\frac{1}{\sqrt{t-3}},\ln(t-2)\rangle$
\end{enumerate}

Let $\mathbf{r}(t)=\langle \cos t,t,\sin t\rangle$ and use it to answer the following questions.

\begin{enumerate}[resume]
\item For what values of $t$ is $\mathbf{r}(t)$ continuous?
\item Sketch the graph of $\mathbf{r}(t)$.
\item Find the domain of $\mathbf{r}(t)=2e^{-t}\,\mathbf{i}+e^{-t}\,\mathbf{j}+\ln(t-1)\,\mathbf{k}$.
\item For what values of $t$ is $\mathbf{r}(t)=2e^{-t}\,\mathbf{i}+e^{-t}\,\mathbf{j}+\ln(t-1)\,\mathbf{k}$ continuous?
\end{enumerate}

Eliminate the parameter $t$, write the equation in Cartesian coordinates, then sketch the graphs of the vector-valued functions.

\begin{enumerate}[resume]
\item $\mathbf{r}(t)=2t\,\mathbf{i}+t^{2}\,\mathbf{j}$ (Hint: Let $x=2t$ and $y=t^{2}$. Solve the first equation for $x$ in terms of $t$ and substitute this result into the second equation.)
\item $\mathbf{r}(t)=t^{3}\,\mathbf{i}+2t\,\mathbf{j}$
\item $\mathbf{r}(t)=2(\sinh t)\,\mathbf{i}+2(\cosh t)\,\mathbf{j},t>0$
\item $\mathbf{r}(t)=3(\cos t)\,\mathbf{i}+3(\sin t)\,\mathbf{j}$
\item $\mathbf{r}(t)=\langle3\sin t,3\cos t\rangle$
\end{enumerate}

Use a graphing utility to sketch each of the following vector-valued functions:

\begin{enumerate}[resume]
\item \techrequired $\mathbf{r}(t)=\left(2\cos^{2}\,t\right)\,\mathbf{i}+(2-\sqrt{t})\,\mathbf{j}$
\item \techrequired $\mathbf{r}(t)=\langle e^{\cos(3t)},e^{-\sin(t)}\rangle$
\item \techrequired $\mathbf{r}(t)=\langle2-\sin(2t),3+2\cos t\rangle$
\item $4x^{2}+9y^{2}=36$; clockwise and counterclockwise
\item $\mathbf{r}(t)=\langle t,t^{2}\rangle$; from left to right
\item The line through $P$ and $Q$ where $P$ is $(1,4,-2)$ and $Q$ is $(3,9,6)$
\end{enumerate}

Consider the curve described by the vector-valued function $\mathbf{r}(t)=(50e^{-t}\cos t)\,\mathbf{i}+(50e^{-t}\sin t)\,\mathbf{j}+(5-5e^{-t})\,\mathbf{k}$.

\begin{enumerate}[resume]
\item What is the initial point of the path corresponding to $\mathbf{r}(0)$?
\item What is $\lim_{t\to \infty}\,\mathbf{r}(t)$?
\item \techrequired Use technology to sketch the curve.
\item Eliminate the parameter $t$ to show that $z=5-\frac{r}{10}$ where $r^{2}=x^{2}+y^{2}$.
\item \techrequired Let $\mathbf{r}(t)=\cos t\,\mathbf{i}+\sin t\,\mathbf{j}+0.3\sin(2t)\,\mathbf{k}$. Use technology to graph the curve (called the roller-coaster curve) over the interval $[0,2\pi)$. Choose at least two views to determine the peaks and valleys.
\item \techrequired Use the result of the preceding problem to construct an equation of a roller coaster with a steep drop from the peak and steep incline from the ``valley.'' Then, use technology to graph the equation.
\item Use the results of the preceding two problems to construct an equation of a path of a roller coaster with more than two turning points (peaks and valleys).
\item
\begin{enumerate}[label=(\alph*)]
  \item Graph the curve $\mathbf{r}(t)=(4+\cos(18t))\cos(t)\,\mathbf{i}+(4+\cos(18t))\sin(t)\,\mathbf{j}+0.3\sin(18t)\,\mathbf{k}$ using two viewing angles of your choice to see the overall shape of the curve.
  \item Does the curve resemble a ``slinky''?
  \item What changes to the equation should be made to increase the number of coils of the slinky?
\end{enumerate}
\end{enumerate}
