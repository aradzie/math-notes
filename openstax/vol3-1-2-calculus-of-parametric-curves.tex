For the following exercises, each set of parametric equations represents a line. Without eliminating the parameter, find the slope of each line.

\begin{enumerate}[start=1]
\item $\begin{matrix}x=3+t, & y=1-t\end{matrix}$
\item $\begin{matrix}x=8+2t, & y=1\end{matrix}$
\item $\begin{matrix}x=4-3t, & y=-2+6t\end{matrix}$
\item $\begin{matrix}x=-5t+7, & y=3t-1\end{matrix}$
\end{enumerate}

For the following exercises, determine the slope of the tangent line, then find an equation of the tangent line at the given value of the parameter.

\begin{enumerate}[resume]
\item $\begin{matrix}x=3\sin t, & y=3\cos t, & t=\frac{\pi}{4}\end{matrix}$
\item $\begin{matrix}x=\cos t, & y=8\sin t, & t=\frac{\pi}{2}\end{matrix}$
\item $\begin{matrix}x=2t, & y=t^{3}, & t=-1\end{matrix}$
\item $\begin{matrix}x=t+\frac{1}{t}, & y=t-\frac{1}{t}, & t=1\end{matrix}$
\item $\begin{matrix}x=\sqrt{t}, & y=2t, & t=4\end{matrix}$
\end{enumerate}

For the following exercises, find all points on the curve that have the given slope.

\begin{enumerate}[resume]
\item $\begin{matrix}x=4\cos t, & y=4\sin t, & \text{slope}=0.5\end{matrix}$
\item $\begin{matrix}x=2\cos t, & y=8\sin t,\,\text{slope}=-1\end{matrix}$
\item $\begin{matrix}x=t+\frac{1}{t}, & y=t-\frac{1}{t},\,\text{slope}=1\end{matrix}$
\item $\begin{matrix}x=2+\sqrt{t}, & y=2-4t,\,\text{slope}=0\end{matrix}$
\end{enumerate}

For the following exercises, write the equation of the tangent line in Cartesian coordinates for the given parameter $t$.

\begin{enumerate}[resume]
\item $\begin{matrix}x=e^{\sqrt{t}}, & y=1-\ln(t^{2}), & t=1\end{matrix}$
\item $\begin{matrix}x=t\ln t, & y=\sin^{2}t, & t=\frac{\pi}{4}\end{matrix}$
\item $x=e^{t}$, $y=(t-1)^{2}$, at $(1,1)$
\item For $x=\sin(2t)$ and $y=2\sin t$, where $0\le t<2\pi$, find all values of $t$ at which a horizontal tangent line exists.
\item For $x=\sin(2t)$ and $y=2\sin t$, where $0\le t<2\pi$, find all values of $t$ at which a vertical tangent line exists.
\item Find all points on the curve $x=4\sin t$, $y=4\cos t$ that have slope $0.5$.
\item Find $\frac{dy}{dx}$ for $x=\sin(t),y=\cos(t)$.
\item Find an equation of the tangent line to $x=\sin(t),y=\cos(t)$ at $t=\frac{\pi}{4}$.
\item For the curve $x=4t$, $y=3t-2$, find the slope and concavity of the curve at $t=3$.
\item For the parametric curve whose equation is $x=4\cos \theta$, $y=4\sin \theta$, find the slope and concavity of the curve at $\theta=\frac{\pi}{4}$.
\item Find the slope and concavity for the curve whose equation is $x=2+\sec \theta$, $y=1+2\tan \theta$ at $\theta=\frac{\pi}{6}$.
\item Find all points on the curve $x=t+4$, $y=t^{3}-3t$ at which there are vertical and horizontal tangents.
\item Find all points on the curve $x=\sec \theta$, $y=\tan \theta$ at which horizontal and vertical tangents exist.
\end{enumerate}

For the following exercises, find $\frac{d^{2}y}{dx^{2}}$.

\begin{enumerate}[resume]
\item $\begin{matrix}x=t^{4}-1, & y=t-t^{2}\end{matrix}$
\item $\begin{matrix}x=\sin(\pi t), & y=\cos(\pi t)\end{matrix}$
\item $\begin{matrix}x=e^{-t}, & y=t\end{matrix}e^{2t}$
\end{enumerate}

For the following exercises, find points on the curve at which the tangent line is horizontal or vertical.

\begin{enumerate}[resume]
\item $\begin{matrix}x=t(t^{2}-3), & y=3(t^{2}-3)\end{matrix}$
\item $\begin{matrix}x=\frac{3t}{1+t^{3}}, & y=\frac{3t^{2}}{1+t^{3}}\end{matrix}$
\end{enumerate}

For the following exercises, find $\frac{dy}{dx}$ at the value of the parameter.

\begin{enumerate}[resume]
\item $\begin{matrix}x=\cos t, & y=\sin t,\,t=\frac{3\pi}{4}\end{matrix}$
\item $\begin{matrix}x=\sqrt{t}, & y=2t+4,\,t=9\end{matrix}$
\item $\begin{matrix}x=4\cos(2\pi s), & y=3\sin(2\pi s), & s=-\frac{1}{4}\end{matrix}$
\end{enumerate}

For the following exercises, find $\frac{d^{2}y}{dx^{2}}$ at the given point without eliminating the parameter.

\begin{enumerate}[resume]
\item $\begin{matrix}x=\frac{1}{2}t^{2}, & y=\frac{1}{3}t^{3}, & t=2\end{matrix}$
\item $x=\sqrt{t}$, $y=2t+4$, $t=1$
\item Find $t$ intervals on which the curve $x=3t^{2}$, $y=t^{3}-t$ is concave up as well as concave down.
\item Determine the concavity of the curve $x=2t+\ln t$, $y=2t-\ln t$.
\item Sketch and find the area under one arch of the cycloid $x=r(\theta-\sin \theta)$, $y=r(1-\cos \theta)$.
\item Find the area bounded by the curve $x=\cos t$, $y=e^{t}$, $0\le t\le \frac{\pi}{2}$ and the lines $y=1$ and $x=0$.
\item Find the area enclosed by the ellipse $x=a\cos \theta$, $y=b\sin \theta$, $0\le \theta<2\pi$.
\item Find the area of the region bounded by $x=2\sin^{2}\theta$, $y=2\sin^{2}\theta\tan \theta$, for $0\le \theta\le \frac{\pi}{2}$.
\end{enumerate}

For the following exercises, find the area of the regions bounded by the parametric curves and the indicated values of the parameter.

\begin{enumerate}[resume]
\item $x=2\cot \theta$, $y=2\sin^{2}\theta$, $0\le \theta\le \pi$
\item \techrequired $x=2a\cos t-a\cos(2t)$, $y=2a\sin t-a\sin(2t)$, $0\le t<2\pi$
\item \techrequired $x=a\sin(2t)$, $y=b\sin t$, $0\le t<2\pi$ (the ``hourglass'')
\item \techrequired $x=2a\cos t-a\sin(2t)$, $y=b\sin t$, $0\le t<2\pi$ (the ``teardrop'')
\end{enumerate}

For the following exercises, find the arc length of the curve on the indicated interval of the parameter.

\begin{enumerate}[resume]
\item $x=4t+3,\,y=3t-2,\,0\le t\le2$
\item $\begin{matrix}x=\frac{1}{3}t^{3}, & y=\frac{1}{2}t^{2}, & 0\le t\le1\end{matrix}$
\item $\begin{matrix}x=\cos(2t), & y=\sin(2t), & 0\le t\le \frac{\pi}{2}\end{matrix}$
\item $\begin{matrix}x=1+t^{2}, & y={(1+t)}^{3}, & 0\le t\le1\end{matrix}$
\item $\begin{matrix}x=e^{t}\cos t, & y=e^{t}\sin t, & 0\le t\le \frac{\pi}{2}\end{matrix}$ (Use a CAS for this and express the answer as a decimal rounded to three places.)
\item $x=a\cos^{3}\theta$, $y=a\sin^{3}\theta$ on the interval $[0,2\pi)$ (the hypocycloid)
\item Find the length of one arch of the cycloid $x=4(t-\sin t)$, $y=4(1-\cos t)$.
\item Find the distance traveled by a particle with position $(x,y)$ as $t$ varies in the given time interval: $\begin{matrix}x=\sin^{2}t, & y=\cos^{2}t, & 0\le t\le3\pi \end{matrix}$.
\item Find the length of one arch of the cycloid $x=\theta-\sin \theta$, $y=1-\cos \theta$.
\item Show that the total length of the ellipse $x=4\sin \theta$, $y=3\cos \theta$ is $L=16\int_{0}^{\pi/2}\sqrt{1-e^{2}\sin^{2}\theta}\,d\theta$, where $e=\frac{c}{a}$ and $c=\sqrt{a^{2}-b^{2}}$.
\item Find the length of the curve $x=e^{t}-t$, $y=4e^{t/2}$, $-8\le t\le3$.
\end{enumerate}

For the following exercises, find the area of the surface obtained by rotating the given curve about the $x$-axis.

\begin{enumerate}[resume]
\item $\begin{matrix}x=t^{3}, & y=t^{2}, & 0\le t\le1\end{matrix}$
\item $\begin{matrix}x=a\cos^{3}\theta, & y=a\sin^{3}\theta, & 0\le \theta\le\frac{\pi}{2}\end{matrix}$
\item \techrequired Use a CAS to find the area of the surface generated by rotating $x=t+t^{3}$, $y=t-\frac{1}{t^{2}}$, $1\le t\le2$ about the $x$-axis. (Answer to three decimal places.)
\item Find the surface area obtained by rotating $x=3t^{2}$, $y=2t^{3}$, $0\le t\le5$ about the $y$-axis.
\item Find the area of the surface generated by revolving $x=t^{2}$, $y=2t$, $0\le t\le4$ about the $x$-axis.
\item Find the surface area generated by revolving $x=t^{2}$, $y=2t^{2}$, $0\le t\le1$ about the $y$-axis.
\end{enumerate}
