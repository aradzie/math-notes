Find a power series solution for the following differential equations.

\begin{enumerate}[start=1]
\item $y''+6y'=0$
\item $5y''+y'=0$
\item $y''+25y=0$
\item $y''-y=0$
\item $2y'+y=0$
\item $y'-2xy=0$
\item $(x-7)y'+2y=0$
\item $y''-xy'-y=0$
\item $(1+x^{2})y''-4xy'+6y=0$
\item $x^{2}y''-xy'-3y=0$
\item $y''-8y'=0$, $y(0)=-2$, and $y'(0)=10$
\item $y''-2xy=0$, $y(0)=1$, and $y'(0)=-3$
\item The differential equation $x^{2}y''+xy'+(x^{2}-1)y=0$ is a Bessel equation of order $1$. Use a power series of the form $y=\sum_{n=0}^{\infty}a_{n}x^{n}$ to find the solution.
\end{enumerate}
