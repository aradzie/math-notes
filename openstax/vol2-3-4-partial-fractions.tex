Express the rational function as a sum or difference of two simpler rational expressions.

\begin{enumerate}[start=1]
\item $\frac{1}{(x-3)(x-2)}$
\item $\frac{x^{2}+1}{x(x+1)(x+2)}$
\item $\frac{1}{x^{3}-x}$
\item $\frac{3x+1}{x^{2}}$
\item $\frac{3x^{2}}{x^{2}+1}$ (Hint: Use long division first.)
\item $\frac{2x^{4}}{x^{2}-2x}$
\item $\frac{1}{(x-1)(x^{2}+1)}$
\item $\frac{1}{x^{2}(x-1)}$
\item $\frac{x}{x^{2}-4}$
\item $\frac{1}{x(x-1)(x-2)(x-3)}$
\item $\frac{1}{x^{4}-1}=\frac{1}{(x+1)(x-1)(x^{2}+1)}$
\item $\frac{3x^{2}}{x^{3}-1}=\frac{3x^{2}}{(x-1)(x^{2}+x+1)}$
\item $\frac{2x}{{(x+2)}^{2}}$
\item $\frac{3x^{4}+x^{3}+20x^{2}+3x+31}{(x+1){(x^{2}+4)}^{2}}$
\end{enumerate}

Use the method of partial fractions to evaluate each of the following integrals.

\begin{enumerate}[resume]
\item $\int \frac{dx}{(x-3)(x-2)}$
\item $\int \frac{3x}{x^{2}+2x-8}\,dx$
\item $\int \frac{dx}{x^{3}-x}$
\item $\int \frac{x}{x^{2}-4}\,dx$
\item $\int \frac{dx}{x(x-1)(x-2)(x-3)}$
\item $\int \frac{2x^{2}+4x+22}{x^{2}+2x+10}\,dx$
\item $\int \frac{dx}{x^{2}-5x+6}$
\item $\int \frac{2-x}{x^{2}+x}\,dx$
\item $\int \frac{2}{x^{2}-x-6}\,dx$
\item $\int \frac{dx}{x^{3}-2x^{2}-4x+8}$
\item $\int \frac{dx}{x^{4}-10x^{2}+9}$
\end{enumerate}

Evaluate the following integrals, which have irreducible quadratic factors.

\begin{enumerate}[resume]
\item $\int \frac{2}{(x-4)(x^{2}+2x+6)}\,dx$
\item $\int \frac{x^{2}}{x^{3}-x^{2}+4x-4}\,dx$
\item $\int \frac{x^{3}+6x^{2}+3x+6}{x^{3}+2x^{2}}\,dx$
\item $\int \frac{x}{(x-1){(x^{2}+2x+2)}^{2}}\,dx$
\end{enumerate}

Use the method of partial fractions to evaluate the following integrals.

\begin{enumerate}[resume]
\item $\int \frac{3x+4}{(x^{2}+4)(3-x)}\,dx$
\item $\int \frac{2}{{(x+2)}^{2}(2-x)}\,dx$
\item $\int \frac{3x+4}{x^{3}-2x-4}\,dx$ (Hint: Use the rational root theorem.)
\end{enumerate}

Use substitution to convert the integrals to integrals of rational functions. Then use partial fractions to evaluate the integrals.

\begin{enumerate}[resume]
\item $\int_{0}^{1}\frac{e^{x}}{36-e^{2x}}\,dx$ (Give the exact answer and the decimal equivalent. Round to five decimal places.)
\item $\int \frac{e^{x}}{e^{2x}-e^{x}}\,dx$
\item $\int \frac{\sin x\,dx}{1-\cos^{2}x}$
\item $\int \frac{\sin x}{\cos^{2}x+\cos x-6}\,dx$
\item $\int \frac{1-\sqrt{x}}{1+\sqrt{x}}\,dx$
\item $\int \frac{dt}{{(e^{t}-e^{-t})}^{2}}$
\item $\int \frac{1+e^{x}}{1-e^{x}}\,dx$
\item $\int \frac{dx}{1+\sqrt{x+1}}$
\item $\int \frac{dx}{\sqrt{x}+\sqrt[4]{x}}$
\item $\int \frac{\cos x}{\sin x(1-\sin x)}\,dx$
\item $\int \frac{e^{x}}{{(e^{2x}-4)}^{2}}\,dx$
\item $\int_{1}^{2}\frac{1}{x^{2}\sqrt{4-x^{2}}}\,dx$
\item $\int \frac{1}{2+e^{-x}}\,dx$
\item $\int \frac{1}{1+e^{x}}\,dx$
\end{enumerate}

Use the given substitution to convert the integral to an integral of a rational function, then evaluate.

\begin{enumerate}[resume]
\item $\int \frac{1}{t-\sqrt[3]{t}}\,dt$; $t=x^{3}$
\item $\int \frac{1}{\sqrt{x}+\sqrt[3]{x}}\,dx$; $x=u^{6}$
\item Graph the curve $y=\frac{x}{1+x}$ over the interval $[0,5]$. Then, find the area of the region bounded by the curve, the $x$-axis, and the line $x=4$. \\ \textit{[Figure: This figure is a graph of the function $y=x/(1+x)$. The graph is only in the first quadrant. It begins at the origin and increases into the first quadrant. The curve stops at $x=5$.]}
\item Find the volume of the solid generated when the region bounded by $y=1/\sqrt{x(3-x)}$, $y=0$, $x=1$, and $x=2$ is revolved about the $x$-axis.
\item The velocity of a particle moving along a line is a function of time given by $v(t)=\frac{88t^{2}}{t^{2}+1}$. Find the distance that the particle has traveled after $t=5$ sec.
\end{enumerate}

Solve the initial-value problem for $x$ as a function of $t$.

\begin{enumerate}[resume]
\item $(t^{2}-7t+12)\frac{dx}{dt}=1$, $t>4$, $x(5)=0$
\item $(t+5)\frac{dx}{dt}=x^{2}+1$, $t>-5$, $x(1)=\tan 1$
\item $(2t^{3}-2t^{2}+t-1)\frac{dx}{dt}=3$, $x(2)=0$
\item Find the $x$-coordinate of the centroid of the area bounded by $y(x^{2}-9)=1$, $y=0$, $x=4$, and $x=5$. (Round the answer to two decimal places.)
\item Find the volume generated by revolving the area bounded by $y=\frac{1}{x^{3}+7x^{2}+6x}$, $x=1$, $x=7$, and $y=0$ about the $y$-axis.
\item Find the area bounded by $y=\frac{x-12}{x^{2}-8x-20}$, $y=0$, $x=2$, and $x=4$. (Round the answer to the nearest hundredth.)
\item Evaluate the integral $\int \frac{dx}{x^{3}+1}$.
\end{enumerate}

For the following problems, use the substitutions $\tan(\frac{x}{2})=t$, $dx=\frac{2}{1+t^{2}}dt$, $\sin x=\frac{2t}{1+t^{2}}$, and $\cos x=\frac{1-t^{2}}{1+t^{2}}$.

\begin{enumerate}[resume]
\item $\int \frac{dx}{3-5\sin x}$
\item Find the area under the curve $y=\frac{1}{1+\sin x}$ between $x=0$ and $x=\pi$. (Assume the dimensions are in inches.)
\item Given $\tan(\frac{x}{2})=t$, derive the formulas $dx=\frac{2}{1+t^{2}}dt$, $\sin x=\frac{2t}{1+t^{2}}$, and $\cos x=\frac{1-t^{2}}{1+t^{2}}$.
\item Evaluate $\int \frac{\sqrt[3]{x-8}}{x}\,dx$.
\end{enumerate}
